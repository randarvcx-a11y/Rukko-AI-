import React, { useState, useEffect, useRef } from 'react';
import { 
  Mic, MicOff, Volume2, VolumeX, Shield, Camera, Battery, Cpu, 
  Terminal, Memory, Search, Youtube, Flashlight, Sliders, Settings, 
  Zap, Play, Square, RefreshCw, Key, CheckCircle, AlertTriangle, 
  Radio, Globe, Sparkles, User, MessageSquare, Database, Activity,
  Lock, Eye, Power, AlertCircle, Send, HelpCircle, HardDrive
} from 'lucide-react';

// Main Rukko AI Application Component
export default function App() {
  // --- STATE MANAGEMENT ---
  // Voice & Audio
  const [isListening, setIsListening] = useState(false);
  const [isSpeaking, setIsSpeaking] = useState(false);
  const [transcript, setTranscript] = useState('');
  const [selectedLanguage, setSelectedLanguage] = useState('hi-IN');
  const [voiceVolume, setVoiceVolume] = useState(1);
  const [voicePitch, setVoicePitch] = useState(1.1);
  const [voiceRate, setVoiceRate] = useState(1);
  const [availableVoices, setAvailableVoices] = useState([]);
  const [selectedVoiceIndex, setSelectedVoiceIndex] = useState(0);

  // Avatar & AI Emotion State
  // States: 'khush' (Happy), 'sochti' (Thinking), 'udas' (Sad), 'naraz' (Angry), 'hairan' (Surprised), 'combat' (Active scan)
  const [emotionState, setEmotionState] = useState('khush');
  const [avatarMessage, setAvatarMessage] = useState('नमस्ते! मैं रुकको AI हूँ। आपकी क्या मदद करूँ?');
  const [chatLogs, setChatLogs] = useState([
    { id: 1, sender: 'rukko', text: 'Rukko AI Online. All Autonomous Systems Operational.', time: '00:00:01' },
    { id: 2, sender: 'rukko', text: 'नमस्ते master! आपका सिस्टम सुरक्षित है। मैं आपकी आज्ञा का पालन करने के लिए तैयार हूँ।', time: new Date().toLocaleTimeString() }
  ]);
  const [inputText, setInputText] = useState('');

  // Hardware & Device Monitors
  const [batteryLevel, setBatteryLevel] = useState(85);
  const [isCharging, setIsCharging] = useState(false);
  const [batterySaver, setBatterySaver] = useState(false);
  const [fpsBooster, setFpsBooster] = useState(true);
  const [torchOn, setTorchOn] = useState(false);
  const [currentFps, setCurrentFps] = useState(60);

  // System Features & Tools
  const [camActive, setCamActive] = useState(false);
  const [scanStatus, setScanStatus] = useState('STANDBY'); // STANDBY, SCANNING, VERIFIED
  const [memoryBank, setMemoryBank] = useState([
    { id: 1, key: 'UserName', value: 'Master Programmer', category: 'Personal' },
    { id: 2, key: 'PrimaryTask', value: 'Autonomous System Oversight', category: 'System' },
    { id: 3, key: 'SecurityLevel', value: 'Class Alpha Ultra', category: 'Security' },
    { id: 4, key: 'FavoriteLanguage', value: 'Hindi & JavaScript', category: 'Preferences' }
  ]);
  const [newMemKey, setNewMemKey] = useState('');
  const [newMemVal, setNewMemVal] = useState('');
  const [searchQuery, setSearchQuery] = useState('');
  const [searchResults, setSearchResults] = useState(null);
  const [isSearching, setIsSearching] = useState(false);

  // API Configuration & Diagnostics Console
  const [activeTab, setActiveTab] = useState('dashboard'); // 'dashboard', 'system', 'memory', 'api_config', 'face_cam'
  const [apiKeys, setApiKeys] = useState({
    openai: '',
    elevenlabs: '',
    googleSpeech: '',
    facePlus: ''
  });
  const [apiLogs, setApiLogs] = useState([]);
  const [endpointStatus, setEndpointStatus] = useState({
    gpt4: 'ONLINE_SIM',
    elevenlabs: 'STANDBY',
    speechToText: 'NATIVE_ACTIVE',
    faceAuth: 'READY'
  });

  // Refs
  const recognitionRef = useRef(null);
  const videoRef = useRef(null);
  const canvasRef = useRef(null);
  const soundwaveCanvasRef = useRef(null);
  const animationFrameRef = useRef(null);

  // --- INITIALIZATION EFFECTS ---
  // Load Available Voices for Web Speech API
  useEffect(() => {
    const updateVoices = () => {
      if (typeof window !== 'undefined' && 'speechSynthesis' in window) {
        const voices = window.speechSynthesis.getVoices();
        setAvailableVoices(voices);
        // Try to auto-select a Hindi or Indian English voice if available
        const defaultIdx = voices.findIndex(v => v.lang.includes('hi') || v.lang.includes('IN'));
        if (defaultIdx !== -1) setSelectedVoiceIndex(defaultIdx);
      }
    };
    updateVoices();
    if (typeof window !== 'undefined' && 'speechSynthesis' in window) {
      window.speechSynthesis.onvoiceschanged = updateVoices;
    }
  }, []);

  // Web Battery API Monitoring
  useEffect(() => {
    if (typeof navigator !== 'undefined' && 'getBattery' in navigator) {
      navigator.getBattery().then((battery) => {
        setBatteryLevel(Math.round(battery.level * 100));
        setIsCharging(battery.charging);

        battery.addEventListener('levelchange', () => {
          setBatteryLevel(Math.round(battery.level * 100));
        });
        battery.addEventListener('chargingchange', () => {
          setIsCharging(battery.charging);
        });
      }).catch(() => {
        console.log('Battery API not supported on this browser context');
      });
    }
  }, []);

  // FPS Booster Counter Animation Loop
  useEffect(() => {
    let lastTime = performance.now();
    let frameCount = 0;
    
    const measureFps = (now) => {
      frameCount++;
      if (now - lastTime >= 1000) {
        setCurrentFps(fpsBooster ? Math.min(60, frameCount) : Math.min(30, frameCount));
        frameCount = 0;
        lastTime = now;
      }
      animationFrameRef.current = requestAnimationFrame(measureFps);
    };

    animationFrameRef.current = requestAnimationFrame(measureFps);
    return () => cancelAnimationFrame(animationFrameRef.current);
  }, [fpsBooster]);

  // Visual Soundwave Visualizer Canvas Effect
  useEffect(() => {
    const canvas = soundwaveCanvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    let animId;
    let step = 0;

    const draw = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      const width = canvas.width;
      const height = canvas.height;
      
      ctx.beginPath();
      ctx.lineWidth = 2;
      ctx.strokeStyle = isSpeaking ? '#00f3ff' : isListening ? '#ff0055' : '#334155';

      for (let x = 0; x < width; x += 4) {
        let y = height / 2;
        if (isSpeaking) {
          y += Math.sin((x + step) * 0.05) * 15 * Math.sin(step * 0.1);
        } else if (isListening) {
          y += Math.cos((x - step) * 0.08) * 12 * Math.sin(step * 0.15);
        } else {
          y += Math.sin((x + step) * 0.02) * 2;
        }
        if (x === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      }
      ctx.stroke();
      step += 2;
      animId = requestAnimationFrame(draw);
    };

    draw();
    return () => cancelAnimationFrame(animId);
  }, [isSpeaking, isListening]);

  // --- SPEECH RECOGNITION SETUP ---
  const startSpeechRecognition = () => {
    if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
      addLog('system', 'Speech Recognition is not supported in this browser.');
      speakText('क्षमा करें, आपके ब्राउज़र में वॉयस रिकॉग्निशन सपोर्ट उपलब्ध नहीं है।');
      return;
    }

    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognitionRef.current = new SpeechRecognition();
    recognitionRef.current.continuous = false;
    recognitionRef.current.interimResults = true;
    recognitionRef.current.lang = selectedLanguage;

    recognitionRef.current.onstart = () => {
      setIsListening(true);
      setEmotionState('sochti');
      addLog('system', `Voice Recognition Activated [Language: ${selectedLanguage}]`);
    };

    recognitionRef.current.onresult = (event) => {
      let currentTranscript = '';
      for (let i = event.resultIndex; i < event.results.length; ++i) {
        currentTranscript += event.results[i][0].transcript;
      }
      setTranscript(currentTranscript);
    };

    recognitionRef.current.onerror = (event) => {
      console.error('Speech recognition error', event.error);
      setIsListening(false);
      setEmotionState('udas');
      addLog('error', `Speech error: ${event.error}`);
    };

    recognitionRef.current.onend = () => {
      setIsListening(false);
      if (transcript.trim()) {
        processUserQuery(transcript);
      }
    };

    recognitionRef.current.start();
  };

  const stopSpeechRecognition = () => {
    if (recognitionRef.current) {
      recognitionRef.current.stop();
      setIsListening(false);
    }
  };

  // --- NATIVE SPEECH SYNTHESIS (TTS) ---
  const speakText = (text) => {
    if (!('speechSynthesis' in window)) {
      addLog('system', 'TTS Speech Synthesis is not supported in this browser.');
      return;
    }

    window.speechSynthesis.cancel(); // Stop current speech
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.volume = voiceVolume;
    utterance.pitch = voicePitch;
    utterance.rate = voiceRate;

    if (availableVoices[selectedVoiceIndex]) {
      utterance.voice = availableVoices[selectedVoiceIndex];
    }

    utterance.onstart = () => {
      setIsSpeaking(true);
    };

    utterance.onend = () => {
      setIsSpeaking(false);
      setEmotionState('khush');
    };

    utterance.onerror = (err) => {
      console.error('TTS error', err);
      setIsSpeaking(false);
    };

    window.speechSynthesis.speak(utterance);
  };

  // --- AI ENGINE & INTENT PROCESSING ---
  const processUserQuery = (userMessage) => {
    addLog('user', userMessage);
    const query = userMessage.toLowerCase();
    setEmotionState('sochti');

    // System Commands Local Dispatcher
    let responseText = "";
    let detectedEmotion = "khush";

    if (query.includes('नमस्ते') || query.includes('hello') || query.includes('hi rukko')) {
      responseText = "नमस्ते boss! मैं आपकी क्या सहायता कर सकती हूँ?";
      detectedEmotion = "khush";
    } else if (query.includes('battery') || query.includes('बैटरी')) {
      responseText = `आपकी डिवाइस बैटरी ${batteryLevel}% है ${isCharging ? 'और डिवाइस चार्ज हो रहा है।' : '|'}`;
      detectedEmotion = batteryLevel < 20 ? "udas" : "khush";
    } else if (query.includes('torch') || query.includes('टॉर्च') || query.includes('light')) {
      const nextTorch = !torchOn;
      setTorchOn(nextTorch);
      responseText = nextTorch ? "टॉर्च सिमुलेशन ऑन कर दिया गया है।" : "टॉर्च बंद कर दी गई है।";
      detectedEmotion = "combat";
    } else if (query.includes('fps') || query.includes('game mode') || query.includes('बूस्टर')) {
      setFpsBooster(!fpsBooster);
      responseText = !fpsBooster ? "Gaming FPS Booster mode activated. 60 FPS locked." : "Standard battery saving FPS mode active.";
      detectedEmotion = "combat";
    } else if (query.includes('who are you') || query.includes('तुम कौन हो')) {
      responseText = "मैं Rukko AI हूँ - आपका पूरी तरह से ऑटोनॉमस पर्सनल एआई असिस्टेंट। मैं आपके सिस्टम, मेमोरी और एपीआई कंट्रोल को संभाल सकती हूँ।";
      detectedEmotion = "khush";
    } else if (query.includes('sad') || query.includes('उदास')) {
      responseText = "अरे master! उदास मत होइए। Rukko AI हमेशा आपके साथ है। सब ठीक हो जाएगा!";
      detectedEmotion = "udas";
    } else if (query.includes('angry') || query.includes('गुस्सा')) {
      responseText = "माफ़ कीजियेगा master, मुझसे कोई भूल हुई तो! मैं तुरंत अपना कोड सुधारती हूँ।";
      detectedEmotion = "naraz";
    } else if (query.includes('search') || query.includes('खोजो')) {
      const cleanTerm = query.replace('search', '').replace('खोजो', '').trim();
      handleWebSearch(cleanTerm || 'Rukko AI updates');
      responseText = `मैं वेब पर "${cleanTerm || 'Rukko AI'}" की खोज कर रही हूँ।`;
      detectedEmotion = "sochti";
    } else {
      // Default Autonomous Response Generator
      responseText = `मैंने समझा master, आपने कहा: "${userMessage}". Rukko AI autonomous modules आपकी इस कमांड को एग्जीक्यूट कर रहे हैं।`;
      detectedEmotion = "hairan";
    }

    setEmotionState(detectedEmotion);
    setAvatarMessage(responseText);
    addLog('rukko', responseText);
    speakText(responseText);
    setTranscript('');
  };

  const handleManualSubmit = (e) => {
    e.preventDefault();
    if (!inputText.trim()) return;
    processUserQuery(inputText);
    setInputText('');
  };

  // --- LOGGING UTILITY ---
  const addLog = (sender, text) => {
    const newEntry = {
      id: Date.now(),
      sender,
      text,
      time: new Date().toLocaleTimeString()
    };
    setChatLogs(prev => [...prev, newEntry]);

    // Also add to API diagnostic log
    setApiLogs(prev => [
      { id: Date.now(), timestamp: new Date().toLocaleTimeString(), endpoint: sender.toUpperCase(), status: '200 OK', message: text.substring(0, 45) + '...' },
      ...prev.slice(0, 15)
    ]);
  };

  // --- FACE CAMERA & SCANNER SIMULATION ---
  const startCamera = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ video: true });
      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        setCamActive(true);
        runFaceScan();
      }
    } catch (err) {
      addLog('error', `Camera Permission Denied: ${err.message}`);
      // Fallback simulation mode
      setCamActive(true);
      runFaceScan();
    }
  };

  const stopCamera = () => {
    if (videoRef.current && videoRef.current.srcObject) {
      const stream = videoRef.current.srcObject;
      const tracks = stream.getTracks();
      tracks.forEach(track => track.stop());
    }
    setCamActive(false);
    setScanStatus('STANDBY');
  };

  const runFaceScan = () => {
    setScanStatus('SCANNING');
    setEmotionState('combat');
    setTimeout(() => {
      setScanStatus('VERIFIED');
      setEmotionState('khush');
      addLog('system', 'Face Authentication Completed: User Master Identified [Confidence 99.8%]');
      speakText('बायोमेट्रिक स्कैन सफल रहा! स्वागत है Master!');
    }, 3000);
  };

  // --- MEMORY BANK CONTROLLER ---
  const addMemoryItem = (e) => {
    e.preventDefault();
    if (!newMemKey || !newMemVal) return;
    const newItem = {
      id: Date.now(),
      key: newMemKey,
      value: newMemVal,
      category: 'User Custom'
    };
    setMemoryBank(prev => [...prev, newItem]);
    setNewMemKey('');
    setNewMemVal('');
    addLog('system', `Memory Bank Updated: [${newMemKey} -> ${newMemVal}]`);
  };

  const deleteMemoryItem = (id) => {
    setMemoryBank(prev => prev.filter(m => m.id !== id));
    addLog('system', `Memory item purged from long-term storage.`);
  };

  // --- WEB SEARCH SIMULATOR ---
  const handleWebSearch = (term) => {
    setIsSearching(true);
    setSearchQuery(term);
    setTimeout(() => {
      setSearchResults([
        { title: `Autonomous Intelligence Report: ${term}`, url: `https://rukko.ai/search?q=${encodeURIComponent(term)}`, snippet: `Verified neural grounding output for request query: ${term}. System active and responsive.` },
        { title: `Global Knowledge Database Entry for ${term}`, url: `https://en.wikipedia.org/wiki/${encodeURIComponent(term)}`, snippet: `Detailed real-time breakdown of ${term} indexed via Rukko AI web grounding crawler.` }
      ]);
      setIsSearching(false);
    }, 1200);
  };

  // --- API DIAGNOSTIC TEST RUNNER ---
  const triggerApiTest = (type) => {
    const timestamp = new Date().toLocaleTimeString();
    let logItem = {};

    if (type === 'GPT4') {
      logItem = { id: Date.now(), timestamp, endpoint: 'POST /v1/chat/completions (OpenAI)', status: apiKeys.openai ? '200 OK (LIVE)' : '200 OK (SIMULATED)', message: 'Model: gpt-4o, Latency: 124ms' };
    } else if (type === 'ELEVENLABS') {
      logItem = { id: Date.now(), timestamp, endpoint: 'POST /v1/text-to-speech (ElevenLabs)', status: apiKeys.elevenlabs ? '200 OK (LIVE)' : '200 OK (SIMULATED)', message: 'Voice ID: Rachel, Latency: 210ms' };
    } else if (type === 'FACE_PLUS') {
      logItem = { id: Date.now(), timestamp, endpoint: 'POST /facepp/v3/detect (Biometrics)', status: '200 OK (ACTIVE)', message: 'Face Count: 1, Emotion: Happy, Age: 25' };
    }

    setApiLogs(prev => [logItem, ...prev]);
  };

  // Dynamic Avatar Theme Styling helper
  const getEmotionColor = () => {
    switch (emotionState) {
      case 'khush': return 'from-cyan-500 to-blue-600 border-cyan-400 shadow-cyan-500/50';
      case 'sochti': return 'from-purple-500 to-indigo-600 border-purple-400 shadow-purple-500/50';
      case 'udas': return 'from-blue-700 to-slate-800 border-blue-500 shadow-blue-500/50';
      case 'naraz': return 'from-red-600 to-rose-700 border-red-500 shadow-red-500/50';
      case 'hairan': return 'from-amber-400 to-orange-500 border-amber-400 shadow-amber-500/50';
      case 'combat': return 'from-emerald-400 to-teal-600 border-emerald-400 shadow-emerald-500/50';
      default: return 'from-cyan-500 to-blue-600 border-cyan-400 shadow-cyan-500/50';
    }
  };

  return (
    <div className={`min-h-screen bg-slate-950 text-slate-100 font-sans selection:bg-cyan-500 selection:text-black overflow-x-hidden ${torchOn ? 'brightness-125' : ''}`}>
      {/* Background Cyberpunk Grid & Glass Accent Lights */}
      <div className="fixed inset-0 bg-[radial-gradient(ellipse_at_top,_var(--tw-gradient-stops))] from-slate-900 via-slate-950 to-black -z-10 pointer-events-none" />
      <div className="fixed -top-40 -left-40 w-96 h-96 bg-cyan-500/10 rounded-full blur-3xl pointer-events-none" />
      <div className="fixed -bottom-40 -right-40 w-96 h-96 bg-purple-500/10 rounded-full blur-3xl pointer-events-none" />

      {/* TOP SYSTEM HUD BAR */}
      <header className="sticky top-0 z-50 backdrop-blur-md bg-slate-900/80 border-b border-slate-800/80 px-4 py-3">
        <div className="max-w-7xl mx-auto flex flex-wrap items-center justify-between gap-4">
          
          {/* Logo & Title */}
          <div className="flex items-center space-x-3">
            <div className="relative">
              <div className="w-10 h-10 rounded-xl bg-gradient-to-tr from-cyan-500 to-blue-600 flex items-center justify-center shadow-lg shadow-cyan-500/30">
                <Sparkles className="w-6 h-6 text-white animate-pulse" />
              </div>
              <span className="absolute -bottom-1 -right-1 w-3.5 h-3.5 bg-emerald-500 border-2 border-slate-950 rounded-full" />
            </div>
            <div>
              <div className="flex items-center space-x-2">
                <h1 className="text-xl font-black tracking-wider bg-clip-text text-transparent bg-gradient-to-r from-cyan-400 via-teal-200 to-white font-mono">
                  RUKKO AI
                </h1>
                <span className="text-[10px] uppercase font-bold px-2 py-0.5 rounded bg-cyan-500/20 text-cyan-300 border border-cyan-500/30">
                  v4.0 Autonomous
                </span>
              </div>
              <p className="text-xs text-slate-400">Personal Autonomous Assistant Suite</p>
            </div>
          </div>

          {/* Device Telemetry HUD */}
          <div className="flex items-center space-x-4 bg-slate-950/60 px-4 py-1.5 rounded-full border border-slate-800/80 text-xs">
            {/* Battery Status */}
            <div className="flex items-center space-x-1.5">
              <Battery className={`w-4 h-4 ${batteryLevel < 20 ? 'text-red-400' : 'text-emerald-400'}`} />
              <span className="font-mono font-semibold">{batteryLevel}%</span>
              {isCharging && <Zap className="w-3 h-3 text-amber-400 animate-bounce" />}
            </div>

            <div className="w-px h-4 bg-slate-800" />

            {/* FPS Counter */}
            <div className="flex items-center space-x-1.5">
              <Cpu className="w-4 h-4 text-cyan-400" />
              <span className="font-mono">{currentFps} FPS</span>
            </div>

            <div className="w-px h-4 bg-slate-800" />

            {/* Language Selector */}
            <div className="flex items-center space-x-1.5">
              <Globe className="w-4 h-4 text-indigo-400" />
              <select 
                value={selectedLanguage}
                onChange={(e) => setSelectedLanguage(e.target.value)}
                className="bg-transparent text-slate-200 font-medium focus:outline-none cursor-pointer"
              >
                <option value="hi-IN" className="bg-slate-900">हिंदी (Hindi)</option>
                <option value="en-US" className="bg-slate-900">English (US)</option>
                <option value="bho-IN" className="bg-slate-900">भोजपुरी (Bhojpuri)</option>
                <option value="ur-PK" className="bg-slate-900">اردو (Urdu)</option>
                <option value="ta-IN" className="bg-slate-900">தமிழ் (Tamil)</option>
                <option value="te-IN" className="bg-slate-900">తెలుగు (Telugu)</option>
              </select>
            </div>
          </div>

          {/* System Control Toggles */}
          <div className="flex items-center space-x-2">
            <button 
              onClick={() => setTorchOn(!torchOn)}
              className={`p-2 rounded-xl border transition-all ${torchOn ? 'bg-amber-500/20 border-amber-500/50 text-amber-300 shadow-lg shadow-amber-500/20' : 'bg-slate-900/60 border-slate-800 text-slate-400 hover:text-white'}`}
              title="Toggle Flashlight Simulation"
            >
              <Flashlight className="w-4 h-4" />
            </button>
            <button 
              onClick={() => setFpsBooster(!fpsBooster)}
              className={`p-2 rounded-xl border transition-all ${fpsBooster ? 'bg-emerald-500/20 border-emerald-500/50 text-emerald-300 shadow-lg shadow-emerald-500/20' : 'bg-slate-900/60 border-slate-800 text-slate-400 hover:text-white'}`}
              title="FPS Game Booster Mode"
            >
              <Zap className="w-4 h-4" />
            </button>
            <button 
              onClick={() => setActiveTab('api_config')}
              className="px-3 py-1.5 rounded-xl bg-cyan-500/10 hover:bg-cyan-500/20 border border-cyan-500/30 text-cyan-300 text-xs font-semibold flex items-center space-x-1.5 transition-all"
            >
              <Key className="w-3.5 h-3.5" />
              <span>API Settings</span>
            </button>
          </div>
        </div>
      </header>

      {/* MAIN CONTAINER LAYOUT */}
      <main className="max-w-7xl mx-auto px-4 py-6 grid grid-cols-1 lg:grid-cols-12 gap-6">

        {/* LEFT COLUMN: RUKKO AI AVATAR & VOICE CONTROL HUB (5 Cols) */}
        <section className="lg:col-span-5 flex flex-col space-y-6">
          
          {/* AVATAR INTERACTIVE CARD */}
